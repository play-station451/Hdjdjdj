#!/bin/bash

PROXY_FILE="proxy-list.txt"
OUTPUT_FILE="working_proxies.txt"
TIMEOUT_SECONDS=5
MAX_JOBS=5

> "$OUTPUT_FILE"
LOCKFILE="/tmp/proxy-check.lock"

if [ ! -f "$PROXY_FILE" ]; then
    echo "Error: Proxy file '$PROXY_FILE' not found!"
    exit 1
fi

check_proxy() {
    local proxy_url="$1"
    echo "Testing: socks5://$proxy_url"
    
    http_code=$(curl -s -x "socks5://$proxy_url" --connect-timeout $TIMEOUT_SECONDS -o /dev/null -w "%{http_code}" "http://httpbin.org/ip")

    if [ "$http_code" == "200" ]; then
        echo "✅ Success: Proxy is working. Saving to file."
        {
            flock -n 200
            echo "$proxy_url" >> "$OUTPUT_FILE"
        } 200>"$LOCKFILE"
    elif [ "$http_code" == "000" ]; then
        echo "❌ Failure: Connection failed or timed out."
    else
        echo "❌ Failure: Connection returned HTTP code $http_code."
    fi
    echo "-----------------------------------"
}

export -f check_proxy PROXY_FILE OUTPUT_FILE TIMEOUT_SECONDS LOCKFILE

job_count=0
while IFS= read -r proxy_url; do
    if [ -n "$proxy_url" ]; then
        ((job_count++))
        check_proxy "$proxy_url" &
        
        while (( job_count >= MAX_JOBS )); do
            wait -n
            ((job_count--))
        done
    fi
done < "$PROXY_FILE"

# Wait for all remaining background jobs to finish
wait

echo "Proxy check complete with $((job_count)) final threads."
echo "Working proxies have been saved to $OUTPUT_FILE."```
